Script Engine

Встроенный движок скриптов main.lua

Полный доступ к внутреннему API клиента. Написание кастомных макросов, логики боя и динамического HUD через единый файл конфигурации main.lua с поддержкой горячей перезагрузки на лету.

Асинхронный событийный цикл EventBus
Безопасная изоляция без прямого падения JVM
main.lua 
-- LunaClient Core Controller [main.lua]
local Luna = require("LunaCore")

Luna.onInit(function()
    Luna.Logger.info("LunaClient v1.0 main.lua successfully linked.")
end)

Luna.registerHook("onPacketReceive", function(packet)
    if packet:getId() == "EntityVelocity" then
        return packet:cancel() -- Safe velocity modifier
    end
end)
